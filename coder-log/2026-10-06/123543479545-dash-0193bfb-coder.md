## Answer

The user wants to replace the `OnModify` trigger of the existing table "LAAI Time Entry" (ID 50150) to ensure that:
1. Rows with Status = Posted are read-only.
2. The posting codeunit can still set Status to Posted on an open row.
3. Other fields cannot be modified after Status is set to Posted.

The current implementation already has a basic check in `OnModify` that prevents modification when the row was already posted (using `xRec.Status`). However, this approach fails because `xRec` holds the same values as `Rec` during a modify operation, so it won't detect if the Status field is being changed from Open to Posted.

To fix this, we need to:
1. Read the stored row using `GetBySystemId` to get the actual previous status.
2. Allow modification of all fields when the current status is not Posted.
3. Prevent modification of any field (including Status) when the stored status is Posted.
4. Allow the posting codeunit to change Status from Open to Posted.

This requires a new trigger implementation that correctly checks the stored row's status rather than `xRec`.

Here is the AL code for the modified table:

// TimeEntry/TimeEntry.Table.al
```al
table 50150 "LAAI Time Entry"
{
    DataClassification = CustomerContent;

    fields
    {
        field(1; "Entry No."; Integer)
        {
            AutoIncrement = true;
        }
        field(2; Customer; Code[20])
        {
        }
        field(3; Billable; Boolean)
        {
        }
        field(4; "Time Type"; Enum "LAAI Time Type")
        {
        }
        field(5; "Active Minutes"; Integer)
        {
        }
        field(6; Date; Date)
        {
        }
        field(7; Description; Text[100])
        {
        }
        field(8; "Chat ID"; Text[50])
        {
        }
        field(9; "Project No."; Code[20])
        {
            TableRelation = Job;

            trigger OnValidate()
            begin
                if Rec."Project No." <> xRec."Project No." then
                    Rec.Validate("Project Task No.", '');
            end;
        }
        field(10; "Project Task No."; Code[20])
        {
            TableRelation = "Job Task"."Job Task No." where("Job No." = field("Project No."));
        }
        field(11; "Work Type"; Code[10])
        {
            TableRelation = "Work Type";
        }
        field(12; Status; Enum "LAAI Time Entry Status")
        {
            InitValue = Open;
        }
        field(13; "Posted Entry No."; Integer)
        {
        }
        field(14; "Resource No."; Code[20])
        {
            TableRelation = Resource;
        }
    }

    keys
    {
        key(PK; "Entry No.")
        {
            Clustered = true;
        }
        key(ChatID; "Chat ID")
        {
            Unique = true;
        }
    }

    trigger OnInsert()
    begin
        if Rec.Status = Rec.Status::Posted then
            Error('You cannot add a row that is already Posted.');
    end;

    trigger OnModify()
    var
        StoredRow: Record "LAAI Time Entry";
    begin
        if StoredRow.GetBySystemId(Rec.SystemId) then
            if StoredRow.Status = StoredRow.Status::Posted then
                Error('You cannot change a row that is Posted.');
    end;

    trigger OnDelete()
    begin
        if Rec.Status = Rec.Status::Posted then
            Error('You cannot delete a row that is Posted.');
    end;
}
```

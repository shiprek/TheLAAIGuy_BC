// TimeEntry/LAAI Time Sheet Entries.Codeunit.al
codeunit 50151 "LAAI Time Sheet Entries"
{
    procedure MarkTimeSheetLinePosted(var TimeEntry: Record "LAAI Time Entry")
    var
        DayRows: Record "LAAI Time Entry";
        TimeSheetHeader: Record "Time Sheet Header";
        TimeSheetLine: Record "Time Sheet Line";
        TimeSheetDetail: Record "Time Sheet Detail";
        JobJnlLine: Record "Job Journal Line";
        JobJnlPostLine: Codeunit "Job Jnl.-Post Line";
        QtyToPost: Decimal;
        JobLedgEntryNo: Integer;
        IsHandled: Boolean;
        NotOnTimeSheetErr: Label 'This entry is not on a time sheet. Run PutOnTimeSheet first.';
        NothingToPostErr: Label 'This entry''s day on its time sheet is already posted.';
        SharedDayErr: Label 'Another open entry points at this time sheet day. Each entry needs its own time sheet line.';
    begin
        OnBeforeMarkTimeSheetLinePosted(TimeEntry, IsHandled);
        if IsHandled then
            exit;
        TimeEntry.Get(TimeEntry."Entry No.");
        TimeEntry.TestField(Status, TimeEntry.Status::Open);
        if not TimeSheetLine.Get(TimeEntry."Time Sheet No.", TimeEntry."Time Sheet Line No.") then
            Error(NotOnTimeSheetErr);
        TimeSheetHeader.Get(TimeSheetLine."Time Sheet No.");
        TimeSheetHeader.TestField("Resource No.", TimeEntry."Resource No.");
        TimeSheetLine.TestField("Job No.", TimeEntry."Project No.");
        TimeSheetLine.TestField("Job Task No.", TimeEntry."Project Task No.");
        if not TimeSheetDetail.Get(
            TimeSheetLine."Time Sheet No.", TimeSheetLine."Line No.", TimeEntry.Date) then
            Error(NotOnTimeSheetErr);
        TimeSheetDetail.TestField(Posted, true);
        DayRows.SetRange("Time Sheet No.", TimeSheetDetail."Time Sheet No.");
        DayRows.SetRange("Time Sheet Line No.", TimeSheetDetail."Line No.");
        DayRows.SetRange(Date, TimeSheetDetail.Date);
        DayRows.SetRange("Resource No.", TimeSheetHeader."Resource No.");
        DayRows.SetRange("Project No.", TimeSheetLine."Job No.");
        DayRows.SetRange("Project Task No.", TimeSheetLine."Job Task No.");
        DayRows.SetRange(Status, DayRows.Status::Open);
        if DayRows.Count() <> 1 then
            Error(SharedDayErr);
        TimeEntry."Posted Entry No." := DayRows."Entry No.";
        TimeEntry.Status := TimeEntry.Status::Posted;
        TimeEntry.Modify(true);
        OnAfterMarkTimeSheetLinePosted(TimeEntry);
    end;

    [IntegrationEvent(false, false)]
    local procedure OnBeforeMarkTimeSheetLinePosted(var TimeEntry: Record "LAAI Time Entry"; var IsHandled: Boolean)
    begin
    end;

    [IntegrationEvent(false, false)]
    local procedure OnAfterMarkTimeSheetLinePosted(var TimeEntry: Record "LAAI Time Entry")
    begin
    end;
}

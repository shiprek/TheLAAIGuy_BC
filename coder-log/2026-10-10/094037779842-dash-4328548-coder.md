## Answer

// LeadExtension/LAAI Customer Ext.TableExt.al
```al
tableextension 50100 "LAAI Customer Ext" extends Customer
{
    fields
    {
        field(50100; "Test Field"; Text[50])
        {
            Caption = 'Test Field';
            DataClassification = CustomerContent;
        }
    }
}
```

// LeadExtension/LAAI Customer Card Ext.PageExt.al
```al
pageextension 50100 "LAAI Customer Card Ext" extends "Customer Card"
{
    layout
    {
        addlast(General)
        {
            field("Test Field"; Rec."Test Field")
            {
                ApplicationArea = All;
                ToolTip = 'Specifies the test field for the customer.';
            }
        }
    }
}
```
